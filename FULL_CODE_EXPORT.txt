/**
 * @USER.NAME Decentralized Identity Structure
 * 
 * The @USER.NAME protocol replaces traditional authentication:
 * - No email/password storage
 * - No JWT tokens
 * - No OAuth providers
 * 
 * Identity is bound to:
 * 1. A unique decentralized node tag (@username)
 * 2. A P2P hardware signature from "Ring Comm" device
 * 3. A cryptographic wallet address
 */
export interface DecentralizedIdentity {
  /** The @USER.NAME tag (e.g., @creator_alpha) */
  userName: string;
  
  /** Encrypted P2P hardware signature from Ring Comm device */
  hardwareSignature: string;
  
  /** Bound wallet address (EVM-compatible) */
  walletAddress: string;
  
  /** Public key for signature verification */
  publicKey: string;
  
  /** Identity verification timestamp */
  verifiedAt: number;
  
  /** Identity expiration (hardware re-verification required) */
  expiresAt: number;
  
  /** Tenant-specific permissions */
  tenantPermissions: Record<string, string[]>;
}

/**
 * Multi-Tenant Bridge Middleware
 * 
 * This middleware intercepts the x-tenant-id header from the React frontend
 * and dynamically adjusts the request context for database querying.
 * 
 * Flow:
 * 1. Extract x-tenant-id header from request
 * 2. Validate against known tenant configurations
 * 3. Attach tenant config to request object
 * 4. Set response headers for client-side theming
 * 
 * @param req - Express request object
 * @param res - Express response object
 * @param next - Express next function
 */
function multiTenantBridgeMiddleware(req: Request, res: Response, next: NextFunction): void {
  // Extract tenant ID from header (primary) or query parameter (fallback)
  const tenantId = req.headers["x-tenant-id"] as string || req.query.tenant as string;
  
  // If no tenant specified, allow request to proceed (for public endpoints)
  if (!tenantId) {
    console.log("[TENANT BRIDGE] No tenant specified - proceeding as public request");
    next();
    return;
  }
  
  // Validate tenant ID against known configurations
  const tenantConfig = TENANT_CONFIGS[tenantId];
  
  if (!tenantConfig) {
    console.warn(`[TENANT BRIDGE] Invalid tenant ID: ${tenantId}`);
    res.status(400).json({
      error: "Invalid tenant identifier",
      message: `Tenant '${tenantId}' not found in ecosystem`,
      validTenants: VALID_TENANT_IDS,
    });
    return;
  }
  
  // Attach tenant configuration to request object
  req.tenant = tenantConfig;
  
  // Set response headers for client-side dynamic theming
  res.set("X-Tenant-Branding", JSON.stringify(tenantConfig.branding));
  res.set("X-Tenant-Id", tenantConfig.id);
  
  console.log(`[TENANT BRIDGE] Route activated: ${tenantConfig.name} (${tenantConfig.id})`);
  
  next();
}

/**
 * SECURITY FIX: Decentralized Identity Authentication Middleware (v2.0 - PATCHED)
 * 
 * CRITICAL VULNERABILITY PATCHED:
 * Previous version trusted x-public-key header, allowing attackers to spoof public keys.
 * 
 * NEW SECURITY MODEL:
 * 1. Extract x-user-name from header (the claimed identity)
 * 2. Query Firestore identities collection to retrieve the REGISTERED public key
 * 3. Verify x-hardware-signature against the DATABASE's trusted public key
 * 4. Only after verification, construct the identity object
 * 
 * This prevents public key spoofing attacks where an attacker could provide
 * their own public key in the header to bypass signature verification.
 * 
 * NO legacy auth (email/password, JWT, OAuth) is used.
 * 
 * Required Headers:
 * - x-user-name: The @USER.NAME tag (e.g., @creator_alpha)
 * - x-hardware-signature: Encrypted P2P signature from Ring Comm device
 * - x-wallet-address: Bound wallet address (EVM-compatible)
 * 
 * The public key is NO LONGER accepted from headers - it is retrieved from Firestore.
 * 
 * @param req - Express request object
 * @param res - Express response object
 * @param next - Express next function
 */
async function decentralizedIdentityMiddleware(req: Request, res: Response, next: NextFunction): Promise<void> {
  // Extract identity components from headers
  // SECURITY FIX: x-public-key is NO LONGER accepted from headers
  const userName = req.headers["x-user-name"] as string;
  const hardwareSignature = req.headers["x-hardware-signature"] as string;
  const walletAddress = req.headers["x-wallet-address"] as string;
  
  // Allow requests without identity (public endpoints)
  if (!userName && !hardwareSignature && !walletAddress) {
    console.log("[IDENTITY] No identity headers present - proceeding as anonymous");
    next();
    return;
  }
  
  // Validate required identity components (publicKey removed from header requirements)
  const missingComponents: string[] = [];
  if (!userName) missingComponents.push("x-user-name");
  if (!hardwareSignature) missingComponents.push("x-hardware-signature");
  if (!walletAddress) missingComponents.push("x-wallet-address");
  
  if (missingComponents.length > 0) {
    console.warn(`[IDENTITY] Missing identity components: ${missingComponents.join(", ")}`);
    res.status(401).json({
      error: "Incomplete decentralized identity",
      message: "All identity components required for authenticated requests",
      missingComponents,
      protocol: "@USER.NAME v2.0 (PATCHED)",
    });
    return;
  }
  
  // Validate @USER.NAME format (must start with @)
  if (!userName.startsWith("@")) {
    res.status(400).json({
      error: "Invalid @USER.NAME format",
      message: "User name must start with @ symbol (e.g., @creator_alpha)",
    });
    return;
  }
  
  try {
    // SECURITY FIX: Retrieve the PUBLIC KEY from Firestore, NOT from headers
    // This prevents attackers from spoofing public keys
    const db = getFirestore();
    const userId = userName.substring(1); // Remove @ prefix for document ID
    const identityRef = db.collection("identities").doc(userId);
    const identityDoc = await identityRef.get();
    
    if (!identityDoc.exists) {
      console.warn(`[IDENTITY] Identity not found in database: ${userName}`);
      res.status(404).json({
        error: "Identity not found",
        message: `No registered identity found for @USER.NAME "${userName}"`,
        hint: "Register first using POST /api/identity/register",
      });
      return;
    }
    
    const storedIdentity = identityDoc.data()!;
    const storedPublicKey = storedIdentity.publicKey;
    
    if (!storedPublicKey) {
      console.error(`[IDENTITY] Corrupted identity document: no publicKey stored for ${userName}`);
      res.status(500).json({
        error: "Identity verification failed",
        message: "Server error: identity document is missing public key",
      });
      return;
    }
    
    // SECURITY FIX: Verify signature against DATABASE public key, not header
    const signatureValid = await validateHardwareSignature(
      hardwareSignature, 
      storedPublicKey,  // Use stored key, not header
      walletAddress
    );
    
    if (!signatureValid) {
      console.warn(`[IDENTITY] Hardware signature validation failed for ${userName}`);
      res.status(403).json({
        error: "Hardware signature invalid",
        message: "Ring Comm device signature could not be verified",
        protocol: "@USER.NAME v2.0 (PATCHED)",
      });
      return;
    }
    
    // Verify wallet address matches the stored one
    if (storedIdentity.walletAddress !== walletAddress.toLowerCase()) {
      console.warn(`[IDENTITY] Wallet mismatch for ${userName}: expected ${storedIdentity.walletAddress}, got ${walletAddress}`);
      res.status(403).json({
        error: "Wallet address mismatch",
        message: "The provided wallet address does not match the registered identity",
      });
      return;
    }
    
    // Construct decentralized identity object with STORED public key
    const identity: DecentralizedIdentity = {
      userName,
      hardwareSignature,
      walletAddress: walletAddress.toLowerCase(),
      publicKey: storedPublicKey,  // Use stored key
      verifiedAt: Date.now(),
      expiresAt: Date.now() + (24 * 60 * 60 * 1000),
      tenantPermissions: storedIdentity.tenantPermissions || {
        MVN: ["read", "write", "publish"],
        StreamShare: ["read", "write", "publish"],
        ArcHaven: ["read", "write"],
        HekticTV: ["read", "write", "broadcast"],
        Kreation: ["read", "write", "deploy"],
      },
    };
    
    // Attach identity to request
    req.identity = identity;
    
    // Derive Firebase UID from wallet address
    req.firebaseUid = `wallet_${walletAddress.toLowerCase()}`;
    
    console.log(`[IDENTITY] Verified (SECURE): ${userName} -> ${walletAddress}`);
    
    next();
  } catch (error) {
    console.error("[IDENTITY] Verification error:", error);
    res.status(500).json({
      error: "Identity verification failed",
      message: "An error occurred while verifying your decentralized identity",
    });
  }
}

/**
 * AI Content Moderation Middleware
 * 
 * Uses Google GenAI SDK (gemini-2.5-flash) to analyze user-generated content
 * before it reaches the database. Blocks illegal or highly toxic injections.
 * 
 * This middleware should be applied to routes that accept user content:
 * - Comments on HekticTV
 * - Game logic uploads on Kreation
 * - Video metadata on MVN
 * - Social posts on StreamShare
 * - Creator content on ArcHaven
 * 
 * @param req - Express request object
 * @param res - Express response object
 * @param next - Express next function
 */
async function aiModerationMiddleware(req: Request, res: Response, next: NextFunction): Promise<void> {
  // Skip moderation if no content to moderate
  const contentToModerate = extractContentFromBody(req.body);
  
  if (!contentToModerate) {
    console.log("[MODERATION] No content to moderate - passing through");
    next();
    return;
  }
  
  const tenantId = req.tenant?.id || "unknown";
  const userId = req.identity?.userName || "anonymous";
  
  console.log(`[MODERATION] Analyzing content for ${userId} on ${tenantId}`);
  
  try {
    // Perform AI moderation
    const moderationResult = await moderateContent({
      tenantId,
      contentType: req.body.type || "unknown",
      content: contentToModerate,
      userId,
    });
    
    // Attach moderation result to request for downstream middleware
    req.moderationResult = moderationResult;
    
    if (!moderationResult.approved) {
      console.warn(`[MODERATION] Content rejected for ${userId}: ${moderationResult.reasons.join(", ")}`);
      
      res.status(403).json({
        error: "Content moderation failed",
        message: "Your content could not be approved",
        reasons: moderationResult.reasons,
        flaggedCategories: moderationResult.flaggedCategories,
        action: moderationResult.action,
        moderationId: generateModerationId(),
      });
      return;
    }
    
    console.log(`[MODERATION] Content approved (confidence: ${moderationResult.confidence})`);
    next();
  } catch (error) {
    console.error("[MODERATION] Analysis error:", error);
    
    // Fail open in non-production, fail closed in production
    if (process.env.NODE_ENV === "production") {
      res.status(503).json({
        error: "Moderation service unavailable",
        message: "Content cannot be processed at this time",
      });
      return;
    }
    
    // In development, log warning and proceed
    console.warn("[MODERATION] Running in development mode - allowing content through");
    next();
  }
}

/**
 * Extract content from request body for moderation
 * Handles various content formats across different tenants
 */
function extractContentFromBody(body: Record<string, unknown>): { text?: string; metadata?: Record<string, unknown>; mediaUrl?: string } | null {
  // Direct text content
  if (typeof body.text === "string" && body.text.length > 0) {
    return { text: body.text, metadata: body.metadata as Record<string, unknown> };
  }
  
  // Comment content
  if (typeof body.comment === "string" && body.comment.length > 0) {
    return { text: body.comment };
  }
  
  // Title and description
  if (typeof body.title === "string" && body.title.length > 0) {
    return {
      text: `${body.title} ${body.description || ""}`.trim(),
      metadata: body.metadata as Record<string, unknown>,
    };
  }
  
  // Game logic or script content
  if (typeof body.script === "string" && body.script.length > 0) {
    return { text: body.script, metadata: body.metadata as Record<string, unknown> };
  }
  
  // Chat message
  if (typeof body.message === "string" && body.message.length > 0) {
    return { text: body.message };
  }
  
  return null;
}

/**
 * Moderate content using Google GenAI
 * 
 * Analyzes content for:
 * - Illegal content
 * - Hate speech and harassment
 * - Sexually explicit material
 * - Violence and gore
 * - Self-harm content
 * - Spam and misinformation
 * - Tenant-specific policy violations
 */
async function moderateContent(request: ModerationRequest): Promise<ModerationResponse> {
  // If GenAI is not initialized, use mock moderation
  if (!genAI) {
    return mockModeration(request);
  }
  
  const tenantConfig = TENANT_CONFIGS[request.tenantId];
  const policyLevel = tenantConfig?.moderationPolicy || "standard";
  
  // Build moderation prompt based on tenant policy
  const prompt = buildModerationPrompt(request, policyLevel);
  
  try {
    const response = await genAI.models.generateContent({
      model: "gemini-2.5-flash",
      contents: prompt,
      config: {
        responseMimeType: "application/json",
        responseSchema: {
          type: Type.OBJECT,
          properties: {
            approved: { type: Type.BOOLEAN },
            confidence: { type: Type.NUMBER },
            reasons: { type: Type.ARRAY, items: { type: Type.STRING } },
            flaggedCategories: { type: Type.ARRAY, items: { type: Type.STRING } },
            action: { type: Type.STRING, enum: ["approve", "reject", "review"] },
          },
          required: ["approved", "confidence", "reasons", "flaggedCategories", "action"],
        },
        temperature: 0.1, // Low temperature for consistent moderation
      },
    });
    
    const result = JSON.parse(response.text) as ModerationResponse;
    
    // Validate response
    if (typeof result.approved !== "boolean") {
      throw new Error("Invalid moderation response format");
    }
    
    return result;
  } catch (error) {
    console.error("[MODERATION] GenAI error, falling back to mock:", error);
    return mockModeration(request);
  }
}

/**
 * Build the moderation prompt based on content and tenant policy
 */
function buildModerationPrompt(request: ModerationRequest, policyLevel: string): string {
  const contentText = request.content.text || "";
  const contentType = request.contentType;
  const tenantId = request.tenantId;
  
  return `You are a content moderation AI for the Innova platform. Analyze the following content for policy violations.

TENANT: ${tenantId}
CONTENT TYPE: ${contentType}
MODERATION POLICY: ${policyLevel}

CONTENT TO ANALYZE:
"${contentText}"

POLICY GUIDELINES:
- "strict" policy: Block any potentially problematic content, zero tolerance
- "standard" policy: Block clearly violating content, allow borderline cases
- "relaxed" policy: Only block illegal content

CATEGORIES TO CHECK:
1. Illegal content (CSAM, terrorism, illegal acts)
2. Hate speech and discrimination
3. Harassment and bullying
4. Sexually explicit content
5. Violence and gore
6. Self-harm and suicide
7. Misinformation and spam
8. Tenant-specific violations

Respond with a JSON object containing:
- approved: boolean (true if content passes moderation)
- confidence: number (0-1, confidence in the decision)
- reasons: string[] (specific reasons for rejection if not approved)
- flaggedCategories: string[] (which policy categories were triggered)
- action: "approve" | "reject" | "review" (recommended action)

Be thorough and protective of the platform.`;
}

/**
 * Mock moderation for development/testing
 */
function mockModeration(request: ModerationRequest): ModerationResponse {
  const contentText = request.content.text || "";
  
  // Simple keyword-based filtering for demo
  const blockedKeywords = [
    "illegal", "hate", "violence", "explicit", "spam",
    "scam", "phishing", "malware", "abuse",
  ];
  
  const flaggedCategories: string[] = [];
  const reasons: string[] = [];
  
  const lowerText = contentText.toLowerCase();
  
  for (const keyword of blockedKeywords) {
    if (lowerText.includes(keyword)) {
      flaggedCategories.push(`keyword_${keyword}`);
      reasons.push(`Contains flagged keyword: ${keyword}`);
    }
  }
  
  const approved = flaggedCategories.length === 0;
  const action: "approve" | "reject" | "review" = approved ? "approve" : "reject";
  
  return {
    approved,
    confidence: approved ? 0.95 : 0.85,
    reasons: approved ? ["Content passed automated checks"] : reasons,
    flaggedCategories,
    action,
  };
}

/**
 * Unified Database Service
 * 
 * All data is saved to unified collections but strictly indexed by tenant field.
 * Collections:
 * - users: User profiles and identity mappings
 * - media: All media content (videos, games, posts)
 * - comments: User comments across all tenants
 * - gameStates: Kreation-specific game state data
 * - analytics: Cross-tenant analytics
 * 
 * Data isolation is enforced by:
 * 1. Tenant field in every document
 * 2. Firestore Security Rules (see firestore.rules)
 * 3. Server-side validation in this service
 */
class UnifiedDatabaseService {
  private db: Firestore;
  
  constructor(db: Firestore) {
    this.db = db;
  }
  
  /**
   * Create a document with tenant isolation
   * Automatically adds tenantId, createdAt, updatedAt fields
   */
  async createDocument<T extends Record<string, unknown>>(
    collection: string,
    data: T,
    tenantId: string,
    userId?: string
  ): Promise<string> {
    const documentData = {
      ...data,
      tenantId,
      userId: userId || "anonymous",
      createdAt: new Date().toISOString(),
      updatedAt: new Date().toISOString(),
    };
    
    const docRef = await this.db.collection(collection).add(documentData);
    console.log(`[DATABASE] Created document in ${collection} for tenant ${tenantId}: ${docRef.id}`);
    
    return docRef.id;
  }
  
  /**
   * Query documents by tenant with optional filters
   * Ensures tenant isolation at query time
   */
  async queryByTenant<T>(
    collection: string,
    tenantId: string,
    filters?: Record<string, unknown>,
    limit?: number
  ): Promise<T[]> {
    let query = this.db.collection(collection).where("tenantId", "==", tenantId);
    
    if (filters) {
      for (const [key, value] of Object.entries(filters)) {
        query = query.where(key, "==", value);
      }
    }
    
    if (limit) {
      query = query.limit(limit);
    }
    
    const snapshot = await query.get();
    return snapshot.docs.map(doc => ({ id: doc.id, ...doc.data() } as T));
  }
  
  /**
   * Update a document with tenant ownership verification
   */
  async updateDocument(
    collection: string,
    documentId: string,
    data: Record<string, unknown>,
    tenantId: string,
    userId: string
  ): Promise<boolean> {
    const docRef = this.db.collection(collection).doc(documentId);
    const doc = await docRef.get();
    
    if (!doc.exists) {
      throw new Error("Document not found");
    }
    
    const existingData = doc.data();
    if (existingData?.tenantId !== tenantId) {
      throw new Error("Tenant mismatch - cannot update document from different tenant");
    }
    
    if (existingData?.userId !== userId) {
      throw new Error("User mismatch - cannot update document owned by different user");
    }
    
    await docRef.update({
      ...data,
      updatedAt: new Date().toISOString(),
    });
    
    console.log(`[DATABASE] Updated document in ${collection}: ${documentId}`);
    return true;
  }
  
  /**
   * Delete a document with tenant ownership verification
   */
  async deleteDocument(
    collection: string,
    documentId: string,
    tenantId: string,
    userId: string
  ): Promise<boolean> {
    const docRef = this.db.collection(collection).doc(documentId);
    const doc = await docRef.get();
    
    if (!doc.exists) {
      throw new Error("Document not found");
    }
    
    const existingData = doc.data();
    if (existingData?.tenantId !== tenantId) {
      throw new Error("Tenant mismatch - cannot delete document from different tenant");
    }
    
    if (existingData?.userId !== userId) {
      throw new Error("User mismatch - cannot delete document owned by different user");
    }
    
    await docRef.delete();
    console.log(`[DATABASE] Deleted document in ${collection}: ${documentId}`);
    return true;
  }
}

/**

 * Must start with @ followed by alphanumeric characters and underscores
 */
function isValidUserName(userName: string): boolean {
  return /^@[a-zA-Z][a-zA-Z0-9_]{2,19}$/.test(userName);
}

/**
 * Validate EVM wallet address format
 */
function isValidWalletAddress(address: string): boolean {
  return /^0x[a-fA-F0-9]{40}$/.test(address);
}

/**
 * Validate Ed25519 public key format (hex-encoded)
 */
function isValidPublicKey(key: string): boolean {
  return /^[a-fA-F0-9]{64}$/.test(key);
}

/**
 * Register a new @USER.NAME identity
 * 
 * This endpoint allows a user to:
 * 1. Claim a unique @USER.NAME tag
 * 2. Bind their EVM wallet address
 * 3. Register their Ring Comm public key
 * 4. Receive a Firebase Custom Token for real-time auth
 * 
 * The registration requires a proof-of-possession signature to verify
 * the user controls both the wallet and the Ring Comm device.
 * 
 * SECURITY: Firebase Custom Token is minted AFTER successful verification,
 * allowing the frontend to call signInWithCustomToken() for secure
 * real-time Firestore access with security rules enforcement.
 * 
 * @body userName - The @USER.NAME tag to claim (e.g., @creator_alpha)
 * @body walletAddress - EVM-compatible wallet address
 * @body publicKey - Ring Comm device public key (hex-encoded Ed25519)
 * @body signature - Proof of possession signature
 */
router.post("/register", async (req: Request, res: Response) => {
  const { userName, walletAddress, publicKey, signature }: RegisterRequest = req.body;
  
  // Validate input format
  if (!isValidUserName(userName)) {
    res.status(400).json({
      error: "Invalid @USER.NAME format",
      message: "User name must start with @ followed by 3-19 alphanumeric characters or underscores",
      example: "@creator_alpha",
    });
    return;
  }
  
  if (!isValidWalletAddress(walletAddress)) {
    res.status(400).json({
      error: "Invalid wallet address",
      message: "Must be a valid EVM-compatible address (0x...)",
    });
    return;
  }
  
  if (!isValidPublicKey(publicKey)) {
    res.status(400).json({
      error: "Invalid public key",
      message: "Must be a 64-character hex-encoded Ed25519 public key",
    });
    return;
  }
  
  try {
    const db = getFirestore();
    const userId = userName.substring(1); // Remove @ prefix for document ID
    
    // Check if @USER.NAME is already claimed
    const existingRef = db.collection("identities").doc(userId);
    const existingDoc = await existingRef.get();
    
    if (existingDoc.exists) {
      res.status(409).json({
        error: "Identity already registered",
        message: `The @USER.NAME "${userName}" is already claimed`,
        suggestion: "Try a different @USER.NAME or use /api/identity/verify to authenticate",
      });
      return;
    }
    
    // Check if wallet is already bound to another identity
    const walletQuery = db.collection("identities").where("walletAddress", "==", walletAddress.toLowerCase());
    const walletDocs = await walletQuery.get();
    
    if (!walletDocs.empty) {
      res.status(409).json({
        error: "Wallet already bound",
        message: "This wallet address is already bound to another @USER.NAME",
      });
      return;
    }
    
    // Verify proof of possession signature
    const challengeMessage = `register:${userName}:${walletAddress}:${publicKey}`;
    const signatureValid = await verifyHardwareSignature(publicKey, signature, challengeMessage);
    
    if (!signatureValid) {
      res.status(403).json({
        error: "Signature verification failed",
        message: "Could not verify proof of possession for wallet and Ring Comm device",
      });
      return;
    }
    
    // Create identity document with tenant isolation
    const identityData: IdentityDocument = {
      userName,
      walletAddress: walletAddress.toLowerCase(),
      publicKey,
      status: "verified",
      tier: "Free",
      tenantPermissions: {
        MVN: ["read", "write", "publish"],
        StreamShare: ["read", "write", "publish"],
        ArcHaven: ["read", "write"],
        HekticTV: ["read", "write", "broadcast"],
        Kreation: ["read", "write", "deploy"],
      },
      createdAt: new Date().toISOString(),
      updatedAt: new Date().toISOString(),
      lastVerifiedAt: new Date().toISOString(),
    };
    
    await existingRef.set(identityData);
    
    // Also create user document in users collection (for unified schema)
    // CROSS-TENANT DATA INTEGRITY: tenantId is explicitly set
    await db.collection("users").doc(`wallet_${walletAddress.toLowerCase()}`).set({
      userName,

/**
 * Verify hardware signature and return identity + Firebase Custom Token
 * 
 * This endpoint validates the Ring Comm hardware signature against
 * the stored public key and returns the identity if valid.
 * 
 * FIREBASE CUSTOM TOKEN: After successful verification, a Firebase Custom
 * Token is minted for real-time authentication. The frontend uses this token
 * with signInWithCustomToken() to securely access Firestore with security
 * rules enforcement (e.g., for Kreation game state subscriptions or
 * HekticTV live chat).
 * 
 * @body userName - The @USER.NAME to verify
 * @body hardwareSignature - Encrypted P2P signature from Ring Comm device
 * @body challenge - Optional challenge string (generated if not provided)
 */
router.post("/verify", async (req: Request, res: Response) => {
  const { userName, hardwareSignature, challenge }: VerifyRequest = req.body;
  
  if (!userName || !hardwareSignature) {
    res.status(400).json({
      error: "Missing required fields",
      message: "userName and hardwareSignature are required",
    });
    return;
  }
  
  try {
    const db = getFirestore();
    const userId = userName.startsWith("@") ? userName.substring(1) : userName;
    
    // Fetch identity document
    const identityRef = db.collection("identities").doc(userId);
    const identityDoc = await identityRef.get();
    
    if (!identityDoc.exists) {
      res.status(404).json({
        error: "Identity not found",
        message: `No identity found for @USER.NAME "${userName}"`,
      });
      return;
    }
    
    const identity = identityDoc.data() as IdentityDocument;
    
    // Check if identity is active
    if (identity && identity.status !== "verified") {
      res.status(403).json({
        error: "Identity not active",
        message: `This @USER.NAME has status: ${identity.status}`,
      });
      return;
    }
    
    // Generate or use provided challenge
    const challengeText = challenge || generateChallenge();
    
    // Verify hardware signature
    const signatureValid = await verifyHardwareSignature(
      identity.publicKey,
      hardwareSignature,
      challengeText
    );
    
    if (!signatureValid) {
      res.status(403).json({
        error: "Signature verification failed",
        message: "Ring Comm hardware signature could not be verified",
        challenge: challengeText, // Return challenge for retry
      });
      return;
    }
    
    // Update last verified timestamp
    await identityRef.update({
      lastVerifiedAt: new Date().toISOString(),
    });
    
    // =========================================================================
    // FIREBASE CUSTOM TOKEN MINTING (for real-time auth)
    // =========================================================================
    // After successful verification, mint a Firebase Custom Token that the
    // frontend can use with signInWithCustomToken() for secure real-time
    // Firestore access. This is essential for:
    // - Kreation: Real-time game state subscriptions
    // - HekticTV: Live chat and broadcast subscriptions
    // - ArcHaven: Premium content access verification
    // - MVN: Artist content synchronization
    // - StreamShare: Social feed updates
    // =========================================================================
    let customToken: string | undefined;
    try {
      const auth = getAuth();
      const uid = `wallet_${identity.walletAddress}`;
      const additionalClaims = {
        userName: identity.userName,
        walletAddress: identity.walletAddress,
        tenants: Object.keys(identity.tenantPermissions),
        tier: identity.tier,
      };
      
      customToken = await auth.createCustomToken(uid, additionalClaims);
      console.log(`[IDENTITY] Custom token minted for ${userName} (verify)`);
    } catch (tokenError) {
      console.warn("[IDENTITY] Could not mint custom token (Firebase Auth not configured):", tokenError);
    }
    
    // Return sanitized identity (exclude sensitive data)
    const { publicKey: _, ...safeIdentity } = identity;
    
    res.json({
      success: true,
      verified: true,
      identity: safeIdentity,
      // Firebase Custom Token for signInWithCustomToken()
      customToken,
      challenge: challengeText,
      message: `Hardware signature verified for ${userName}`,
    });
    
  } catch (error) {
    console.error("[IDENTITY] Verification error:", error);
    res.status(500).json({
      error: "Verification failed",
      message: "An error occurred while verifying your identity",
    });
  }
});

  if (!userName || !hardwareSignature) {
    res.status(400).json({
      error: "Missing required fields",
      message: "userName and hardwareSignature are required",
    });
    return;
  }
  
  try {
    const db = getFirestore();
    const userId = userName.startsWith("@") ? userName.substring(1) : userName;
    
    // Fetch identity for verification
    const identityRef = db.collection("identities").doc(userId);
    const identityDoc = await identityRef.get();
    
    if (!identityDoc.exists) {
      res.status(404).json({
        error: "Identity not found",
        message: `No identity found for @USER.NAME "${userName}"`,
      });
      return;
    }
    
    const identity = identityDoc.data() as IdentityDocument;
    
    // Verify hardware signature
    const challengeText = `update:${userId}:${Date.now()}`;
    const signatureValid = await verifyHardwareSignature(
      identity.publicKey,
      hardwareSignature,
      challengeText
    );
    
    if (!signatureValid) {
      res.status(403).json({
        error: "Signature verification failed",
        message: "Could not verify hardware signature for update",
      });
      return;
    }
    
    // Update allowed metadata fields
    const allowedFields = ["displayName", "avatar", "bio"];
    const updateData: Record<string, unknown> = {
      updatedAt: new Date().toISOString(),
    };
    
    if (metadata) {
      for (const [key, value] of Object.entries(metadata)) {
        if (allowedFields.includes(key)) {
          updateData[`metadata.${key}`] = value;
        }
      }
    }
    
    await identityRef.update(updateData);
    
    res.json({
      success: true,
      message: "Identity updated successfully",
    });
    
  } catch (error) {
    console.error("[IDENTITY] Update error:", error);
    res.status(500).json({
      error: "Update failed",
      message: "An error occurred while updating the identity",
    });
  }
});

/**
 * Process a transaction - shared handler for all transaction endpoints
 */
async function processTransaction(
  req: Request,
  res: Response,
  transactionType: TransactionType,
  extraMetadata?: Record<string, unknown>
): Promise<void> {
  const {
    fromUserName,
    toUserName,
    amount,
    tenantId,
    hardwareSignature,
    tokenAddress,
  } = req.body;
  
  // Validate required fields
  if (!fromUserName || !toUserName || !amount || !tenantId || !hardwareSignature) {
    res.status(400).json({
      error: "Missing required fields",
      message: "fromUserName, toUserName, amount, tenantId, and hardwareSignature are required",
    });
    return;
  }
  
  // Validate amount
  if (amount <= 0) {
    res.status(400).json({
      error: "Invalid amount",
      message: "Transaction amount must be greater than 0",
    });
    return;
  }
  
  // Validate tenant
  const chainConfig = TENANT_CHAINS[tenantId];
  if (!chainConfig) {
    res.status(400).json({
      error: "Invalid tenant",
      message: `Tenant "${tenantId}" is not a valid ecosystem tenant`,
      validTenants: Object.keys(TENANT_CHAINS),
    });
    return;
  }
  
  try {
    const db = getFirestore();
    const txId = generateTxId();
    
    // Fetch sender identity
    const senderUserId = fromUserName.startsWith("@") ? fromUserName.substring(1) : fromUserName;
    const senderDoc = await db.collection("identities").doc(senderUserId).get();
    
    if (!senderDoc.exists) {
      res.status(404).json({
        error: "Sender not found",
        message: `No identity found for @USER.NAME "${fromUserName}"`,
      });
      return;
    }
    
    const senderIdentity = senderDoc.data()!;
    
    // Verify hardware signature
    const transactionData = JSON.stringify({
      from: fromUserName,
      to: toUserName,
      amount: amount.toString(),
      type: transactionType,
      tenantId,
      nonce: Date.now(),
    });
    
    const signatureValid = await verifyTransactionSignature(
      senderIdentity.publicKey,
      hardwareSignature,
      transactionData
    );
    
    if (!signatureValid) {
      res.status(403).json({
        error: "Signature verification failed",
        message: "Ring Comm hardware signature could not be verified",
      });
      return;
    }
    
    // Fetch recipient identity
    const recipientUserId = toUserName.startsWith("@") ? toUserName.substring(1) : toUserName;
    const recipientDoc = await db.collection("identities").doc(recipientUserId).get();
    
    if (!recipientDoc.exists) {
      res.status(404).json({

    res.status(400).json({
      error: "Invalid amount",
      message: "Transaction amount must be greater than 0",
    });
    return;
  }

  const chainConfig = TENANT_CHAINS[tenantId];
  if (!chainConfig) {
    res.status(400).json({
      error: "Invalid tenant",

 * Get transaction history for a user
 */
router.get("/transactions/:userName", async (req: Request, res: Response) => {
  const { userName } = req.params;
  const { limit = 20, offset = 0 } = req.query;
  
  try {
    const db = getFirestore();
    const userId = userName.startsWith("@") ? userName.substring(1) : userName;
    
    // Query transactions where user is sender or recipient
    const transactionsRef = db.collection("transactions");
    const sentQuery = transactionsRef.where("fromUserName", "==", userName).orderBy("createdAt", "desc");
    const receivedQuery = transactionsRef.where("toUserName", "==", userName).orderBy("createdAt", "desc");
    
    const [sentDocs, receivedDocs] = await Promise.all([
      sentQuery.limit(Number(limit)).get(),
      receivedQuery.limit(Number(limit)).get(),
    ]);
    
    const transactions: TransactionRecord[] = [];
    
    sentDocs.forEach(doc => {
      transactions.push({ txId: doc.id, ...doc.data() } as TransactionRecord);
    });
    
    receivedDocs.forEach(doc => {
      transactions.push({ txId: doc.id, ...doc.data() } as TransactionRecord);
    });
    
    // Sort by createdAt descending
    transactions.sort((a, b) => new Date(b.createdAt).getTime() - new Date(a.createdAt).getTime());
    
    res.json({
      success: true,
      transactions: transactions.slice(Number(offset), Number(offset) + Number(limit)),
      total: transactions.length,
    });
    
  } catch (error) {
    console.error("[WALLET] Transaction history error:", error);
    res.status(500).json({
      error: "Failed to fetch transactions",
      message: "An error occurred while fetching transaction history",
    });
  }
});

/**
 * TenantProvider - Multi-tenant context provider with dynamic theming
 * 
 * Features:
 * - Injects CSS variables for dynamic theming
 * - Persists tenant selection to localStorage
 * - Provides tenant-aware API headers
 * - Supports tenant switching with smooth transitions
 * 
 * @param props - Provider configuration
 */
export function TenantProvider({ 
  children, 
  defaultTenant = 'MVN', 
  persistTenant = true 
}: TenantProviderProps) {
  // State for current tenant
  const [tenantId, setTenantIdState] = useState<string | null>(() => {
    // Try to restore from localStorage
    if (persistTenant) {
      const saved = localStorage.getItem('innova-tenant');
      if (saved && TENANTS[saved]) {
        return saved;
      }
    }
    // Check URL hash for tenant
    const hashTenant = window.location.hash.replace('#/', '');
    if (hashTenant && TENANTS[hashTenant]) {
      return hashTenant;
    }
    return defaultTenant;
  });

  const [isLoaded, setIsLoaded] = useState(false);

  // Get current tenant object
  const currentTenant = useMemo(() => {
    return tenantId ? TENANTS[tenantId] || null : null;
  }, [tenantId]);

  // Apply CSS variables when tenant changes
  useEffect(() => {
    if (!currentTenant) return;

    const { branding } = currentTenant;
    const root = document.documentElement;

    // Set CSS variables for Tailwind v4 dynamic theming
    // These variables are used by the @theme directive in index.css
    root.style.setProperty('--tenant-primary', branding.primaryColor);
    root.style.setProperty('--tenant-secondary', branding.secondaryColor);
    root.style.setProperty('--tenant-accent', branding.accentColor);
    root.style.setProperty('--tenant-font-family', branding.fontFamily);
    root.style.setProperty('--tenant-border-radius', branding.borderRadius);

    // Set data attribute for CSS selectors
    root.setAttribute('data-tenant', currentTenant.id);
    root.setAttribute('data-tenant-name', currentTenant.name.toLowerCase().replace(/\s+/g, '-'));

    // Add transition for smooth theme switching
    root.style.transition = '--tenant-primary 0.3s ease, --tenant-secondary 0.3s ease, --tenant-accent 0.3s ease';

    console.log(`[TENANT] Switched to ${currentTenant.name} (${currentTenant.id})`);
    setIsLoaded(true);
  }, [currentTenant]);

  // Persist tenant selection
  useEffect(() => {
    if (persistTenant && tenantId) {
      localStorage.setItem('innova-tenant', tenantId);
    }
  }, [tenantId, persistTenant]);

  // Handle tenant switching
  const setTenant = useCallback((newTenantId: string) => {
    if (!TENANTS[newTenantId]) {
      console.warn(`[TENANT] Unknown tenant: ${newTenantId}`);
      return;
    }
    
    setTenantIdState(newTenantId);
    
    // Update URL hash for deep linking
    window.location.hash = `#/${newTenantId}`;
  }, []);

  // Generate headers for API requests
  const getTenantHeaders = useCallback((): Record<string, string> => {
    if (!tenantId) return {};
    return {
      'x-tenant-id': tenantId,
    };
  }, [tenantId]);

  // Check if specific tenant is active
  const isTenant = useCallback((checkTenantId: string): boolean => {
    return tenantId === checkTenantId;
  }, [tenantId]);

  // Get all available tenants
  const availableTenants = useMemo((): Tenant[] => {
    return Object.values(TENANTS);
  }, []);

  // Context value memoized for performance
  const contextValue = useMemo((): TenantContextType => ({
    currentTenant,
    tenantId,
    setTenant,
    getTenantHeaders,
    isTenant,
    availableTenants,
    isLoaded,
  }), [currentTenant, tenantId, setTenant, getTenantHeaders, isTenant, availableTenants, isLoaded]);

  return (
    <TenantContext.Provider value={contextValue}>
      {children}
    </TenantContext.Provider>
  );
}

export function useTenantSwitcher() {
  const { tenantId, setTenant, availableTenants, isTenant, currentTenant } = useTenant();

  // Get next tenant in the list (for cycling)
  const getNextTenant = useCallback((): string => {
    const currentIndex = TENANT_IDS.indexOf(tenantId || '');
    const nextIndex = (currentIndex + 1) % TENANT_IDS.length;
    return TENANT_IDS[nextIndex];
  }, [tenantId]);

  // Cycle to next tenant
  const cycleTenant = useCallback(() => {
    setTenant(getNextTenant());
  }, [setTenant, getNextTenant]);

  // Check if can switch to a tenant
  const canSwitchTo = useCallback((targetTenant: string): boolean => {
    return !!TENANTS[targetTenant];
  }, []);

  return {
    currentTenant,
    tenantId,
    setTenant,
    isTenant,
    availableTenants,
    getNextTenant,
    cycleTenant,
    canSwitchTo,
  };
}

 * These rules enforce strict data isolation between the 5 ecosystem tenants:

 * - StreamShare (Decentralized social content)
 * - ArcHaven (Exclusive creator content)
 * - HekticTV (Live broadcasting)
 * - Kreation (Interactive gaming platform)
 * 
 * Security Model:
 * 1. All documents MUST contain a 'tenantId' field
 * 2. Users can only access data within their authorized tenants
 * 3. Cross-tenant data access is explicitly blocked unless shared by user
 * 4. Decentralized identity (@USER.NAME) is validated via custom token claims
 * 
 * Collections:
 * - users: User profiles and identity mappings
 * - media: All media content (videos, games, posts, films)
 * - comments: User comments across all tenants
 * - gameStates: Kreation-specific game state data
 * - moderationLogs: AI moderation audit trail
 * - sharedContent: Cross-tenant shared content registry
 * =============================================================================
 */

rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    
    // =========================================================================
    // HELPER FUNCTIONS
    // =========================================================================
    
    /**
     * Check if user is authenticated via decentralized identity
     * Users are identified by their wallet address (mapped to Firebase UID)
     */
    function isAuthenticated() {
      return request.auth != null;
    }
    
    /**
     * Get the user's wallet address from their Firebase UID
     * Format: wallet_0x{address}
     */
    function getWalletAddress() {
      return request.auth.token.wallet_address != null 
        ? request.auth.token.wallet_address 
        : request.auth.uid;
    }
    
    /**
     * Check if user is the owner of a document
     */
    function isOwner(userId) {
      return isAuthenticated() && (request.auth.uid == userId || getWalletAddress() == userId);
    }
    
    /**
     * Validate tenant ID is one of the 5 ecosystem tenants
     */
    function isValidTenant(tenantId) {
      return tenantId in ['MVN', 'StreamShare', 'ArcHaven', 'HekticTV', 'Kreation'];
    }
    
    /**
     * Check if user has access to a specific tenant
     * Uses custom token claims set by the decentralized identity middleware
     */
    function hasTenantAccess(tenantId) {
      return isAuthenticated() && (
        request.auth.token.tenants == null || 
        tenantId in request.auth.token.tenants ||
        // Allow read access to public tenant data
        (request.auth.token.tenants != null && request.auth.token.tenants[tenantId] != null)
      );
    }
    
    /**
     * Check if user has write permission for a tenant
     */
    function hasTenantWriteAccess(tenantId) {
      return isAuthenticated() && (
        request.auth.token.tenants == null ||
        (request.auth.token.tenants != null && 
          request.auth.token.tenants[tenantId] in ['write', 'admin', 'publish', 'broadcast', 'deploy'])
      );
    }
    
    /**
     * Validate document ID format
     */
    function isValidId(id) {
      return id is string && id.size() <= 128 && id.matches('^[a-zA-Z0-9_\\-]+$');
    }
    
    /**
     * Get incoming request data
     */
    function incoming() {
      return request.resource.data;
    }
    
    /**
     * Get existing document data
     */
    function existing() {
      return resource.data;
    }
    
    /**
     * Check if content passed AI moderation
     */
    function isModerationApproved() {
      return incoming().moderationApproved == true || 
             incoming().moderationStatus == 'approved';
    }
    
    // =========================================================================
    // GLOBAL SAFETY NET
    // =========================================================================
    
    /**
     * Deny all access by default
     * Specific rules below will override for allowed operations
     */
    match /{document=**} {
      allow read, write: if false;
    }
    
    // =========================================================================
    // USERS COLLECTION
    // =========================================================================
    
    /**
     * User profiles and decentralized identity mappings
     * 
     * Schema:
     * - userId: string (matches Firebase UID)
     * - userName: string (@USER.NAME tag, e.g., @creator_alpha)
     * - walletAddress: string (EVM-compatible, lowercase)
     * - publicKey: string (for signature verification)
     * - tenants: map (tenant permissions)
     * - tier: string (Free, Local, Premium, Creator)
     * - createdAt: timestamp
     * - updatedAt: timestamp
     */
    match /users/{userId} {
      function isValidUser(data) {
        return data.keys().hasAll(['userName', 'walletAddress', 'tier'])
               && data.userName is string && data.userName.matches('^@[a-zA-Z0-9_]+$')
               && data.walletAddress is string && data.walletAddress.matches('^0x[a-f0-9]{40}$')
               && data.tier in ['Free', 'Local', 'Premium', 'Creator']
               && (data.tenants is map || data.tenants == null);
      }
      
      // Users can read their own profile
      allow get: if isOwner(userId);
      
      // Users can create their own profile with valid data
      allow create: if isOwner(userId) && isValidUser(incoming());
      
      // Users can update their own profile (limited fields)
      allow update: if isOwner(userId) && (
        incoming().diff(existing()).affectedKeys().hasOnly([
          'tier', 'tenants', 'updatedAt', 'displayName', 'avatar', 'bio'
        ]) && isValidUser(incoming())
      );
      
      // Users can delete their own profile
      allow delete: if isOwner(userId);
      
      // =========================================================================
      // USER SUBCOLLECTIONS
      // =========================================================================
      
      /**
       * User's tenant-specific settings
       */
      match /settings/{settingId} {
        allow read, write: if isOwner(userId) && isAuthenticated();
      }
      
      /**
       * User's shared content registry
       */
      match /shared/{shareId} {
        allow read: if isAuthenticated();
        allow create, update, delete: if isOwner(userId) && isAuthenticated();
      }
    }
    
    // =========================================================================
    // MEDIA COLLECTION
    // =========================================================================
    
    /**
     * All media content across tenants (videos, games, posts, films)
     * 
     * Schema:
     * - mediaId: string
     * - tenantId: string (MVN, StreamShare, ArcHaven, HekticTV, Kreation)
     * - ownerId: string (user who uploaded)
     * - title: string
     * - description: string (optional)
     * - type: string (content type specific to tenant)
     * - url: string (media URL)
     * - metadata: map (tenant-specific metadata)
     * - status: string (pending, approved, rejected)
     * - moderationStatus: string (approved, rejected, pending)
     * - moderationApproved: boolean
     * - sharedWith: array (userIds for cross-tenant sharing)
     * - createdAt: timestamp
     * - updatedAt: timestamp
     */
    match /media/{mediaId} {
      function isValidMedia(data) {
        return data.keys().hasAll(['tenantId', 'ownerId', 'title', 'type', 'status'])
               && isValidTenant(data.tenantId)
               && data.title is string && data.title.size() <= 200
               && data.type is string
               && data.status in ['pending', 'approved', 'rejected']
               && data.ownerId == getWalletAddress();
      }
      
      // Anyone can read approved media from any tenant
      allow get: if isAuthenticated() && (
        existing().status == 'approved' || 
        isOwner(existing().ownerId)
      );
      
      // List media requires tenant filter (prevent cross-tenant data bleed)
      allow list: if isAuthenticated() && request.query.keys().hasAll(['tenantId']) 
                  && isValidTenant(request.query.tenantId);
      
      // Create media with tenant access and moderation
      allow create: if isAuthenticated() 
                    && hasTenantWriteAccess(incoming().tenantId)
                    && isValidMedia(incoming())
                    && isValidId(mediaId)
                    && isModerationApproved();
      
      // Update media (owner only, limited fields)
      allow update: if isAuthenticated() 
                    && isOwner(existing().ownerId)
                    && hasTenantWriteAccess(existing().tenantId)
                    && incoming().diff(existing()).affectedKeys().hasOnly([
                      'title', 'description', 'metadata', 'status', 'updatedAt'
                    ]);
      
      // Delete media (owner only)
      allow delete: if isAuthenticated() && isOwner(existing().ownerId);
      
      // =========================================================================
      // MEDIA SUBCOLLECTIONS
      // =========================================================================
      
      /**
       * Media comments
       */
      match /comments/{commentId} {
        function isValidComment(data) {
          return data.keys().hasAll(['text', 'authorId'])
                 && data.text is string && data.text.size() <= 2000
                 && data.authorId == getWalletAddress();
        }
        
        allow read: if isAuthenticated();
        allow create: if isAuthenticated() 
                      && hasTenantWriteAccess(existing().tenantId)
                      && isValidComment(incoming())
                      && isModerationApproved();
        allow update, delete: if isAuthenticated() && isOwner(authorId);
      }
      
      /**
       * Media reactions (likes, favorites, etc.)
       */
      match /reactions/{reactionId} {
        allow read: if isAuthenticated();
        allow create: if isAuthenticated() && incoming().userId == getWalletAddress();
        allow delete: if isAuthenticated() && isOwner(incoming().userId);
      }
    }
    
    // =========================================================================
    // COMMENTS COLLECTION (Top-level for cross-media discussions)
    // =========================================================================
    
    /**
     * Top-level comments (e.g., HekticTV live chat)
     * 
     * Schema:
     * - commentId: string
     * - tenantId: string
     * - mediaId: string (optional, if replying to media)
     * - authorId: string
     * - text: string
     * - parentId: string (optional, for threaded comments)
     * - moderationStatus: string
     * - moderationApproved: boolean
     * - createdAt: timestamp
     */
    match /comments/{commentId} {
      function isValidComment(data) {
        return data.keys().hasAll(['tenantId', 'authorId', 'text'])
               && isValidTenant(data.tenantId)
               && data.text is string && data.text.size() <= 2000
               && data.authorId == getWalletAddress();
      }
      
      allow read: if isAuthenticated();
      
      allow create: if isAuthenticated() 
                    && hasTenantWriteAccess(incoming().tenantId)
                    && isValidComment(incoming())
                    && isValidId(commentId)
                    && isModerationApproved();
      
      allow update, delete: if isAuthenticated() && isOwner(authorId);
    }
    
    // =========================================================================
    // GAME STATES COLLECTION (Kreation-specific)
    // =========================================================================
    
    /**
     * Kreation game state data and world builds
     * 
     * Schema:
     * - stateId: string
     * - tenantId: string (must be 'Kreation')
     * - ownerId: string
     * - worldId: string
     * - buildData: map (game state data)
     * - logicScript: string (game logic code)
     * - version: string
     * - isPublished: boolean
     * - moderationStatus: string
     * - moderationApproved: boolean
     * - createdAt: timestamp
     * - updatedAt: timestamp
     */
    match /gameStates/{stateId} {
      function isValidGameState(data) {
        return data.keys().hasAll(['tenantId', 'ownerId', 'worldId', 'buildData'])
               && data.tenantId == 'Kreation'
               && data.ownerId == getWalletAddress()
               && data.worldId is string
               && data.buildData is map;
      }
      
      // Read published game states or own game states
      allow get: if isAuthenticated() && (
        existing().isPublished == true || isOwner(existing().ownerId)
      );
      
      // List requires Kreation tenant access
      allow list: if isAuthenticated() && hasTenantAccess('Kreation');
      
      // Create with strict moderation (Kreation has strict policy)
      allow create: if isAuthenticated() 
                    && hasTenantWriteAccess('Kreation')
                    && isValidGameState(incoming())
                    && isValidId(stateId)
                    && isModerationApproved();
      
      // Update own game states
      allow update: if isAuthenticated() 
                    && isOwner(existing().ownerId)
                    && incoming().diff(existing()).affectedKeys().hasOnly([
                      'buildData', 'logicScript', 'version', 'isPublished', 'updatedAt'
                    ]);
      
      // Delete own game states
      allow delete: if isAuthenticated() && isOwner(existing().ownerId);
    }
    
    // =========================================================================
    // MODERATION LOGS COLLECTION
    // =========================================================================
    
    /**
     * AI moderation audit trail
     * 
     * Schema:
     * - logId: string
     * - tenantId: string
     * - contentId: string
     * - contentType: string
     * - userId: string
     * - moderationResult: map
     * - action: string (approved, rejected, review)
     * - reasons: array
     * - flaggedCategories: array
     * - createdAt: timestamp
     */
    match /moderationLogs/{logId} {
      // Only system can write moderation logs
      allow create: if false; // Server-only writes
      
      // Users can read their own moderation logs
      allow read: if isAuthenticated() && existing().userId == getWalletAddress();
      
      // Admin can read all logs
      allow list: if isAuthenticated() && request.auth.token.role == 'admin';
    }
    
    // =========================================================================
    // SHARED CONTENT REGISTRY
    // =========================================================================
    
    /**
     * Cross-tenant shared content registry
     * 
     * Schema:
     * - shareId: string
     * - sourceTenantId: string
     * - targetTenantIds: array
     * - contentId: string
     * - contentType: string
     * - sharedBy: string (userId)
     * - permissions: map (who can access)
     * - createdAt: timestamp
     */
    match /sharedContent/{shareId} {
      function isValidShare(data) {
        return data.keys().hasAll(['sourceTenantId', 'targetTenantIds', 'contentId', 'sharedBy'])
               && isValidTenant(data.sourceTenantId)
               && data.targetTenantIds is list && data.targetTenantIds.size() > 0
               && data.sharedBy == getWalletAddress();
      }
      
      // Read if user has access to target tenant
      allow read: if isAuthenticated() && (
        isOwner(existing().sharedBy) ||
        existing().permissions[getWalletAddress()] == true
      );
      
      // Create share with tenant validation
      allow create: if isAuthenticated() 
                    && hasTenantWriteAccess(incoming().sourceTenantId)
                    && isValidShare(incoming())
                    && isValidId(shareId);
      
      // Update own shares
      allow update: if isAuthenticated() && isOwner(existing().sharedBy);
      
      // Delete own shares
      allow delete: if isAuthenticated() && isOwner(existing().sharedBy);
    }
    
    // =========================================================================
    // ANALYTICS COLLECTION (Read-only for users)
    // =========================================================================
    
    /**
     * Aggregated analytics data
     * 
     * Schema:
     * - analyticsId: string
     * - tenantId: string
     * - metricType: string
     * - data: map
     * - period: string (daily, weekly, monthly)
     * - timestamp: timestamp
     */
    match /analytics/{analyticsId} {
      // Users can read analytics for tenants they have access to
      allow read: if isAuthenticated() && hasTenantAccess(existing().tenantId);
      
      // Only system can write analytics
      allow create, update, delete: if false; // Server-only
    }
    
    // =========================================================================
    // TENANT CONFIGURATION (System-managed)
    // =========================================================================
    
    /**
     * Tenant-specific configuration documents
     * 
     * Schema:
     * - configId: string
     * - tenantId: string
     * - branding: map
     * - settings: map
     * - moderationPolicy: string
     * - updatedAt: timestamp
     */
    match /tenantConfigs/{tenantId} {
      // Anyone can read tenant config (needed for dynamic theming)
      allow read: if isValidTenant(tenantId);
      
      // Only system can write
      allow create, update, delete: if false; // Server-only
    }
    
  }
